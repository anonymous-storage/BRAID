cmake_minimum_required(VERSION 3.19)
project(braid_c_consumer LANGUAGES C)
find_package(BRAID 0.1 CONFIG REQUIRED)
add_executable(consumer ../../examples/encode.c)
target_link_libraries(consumer PRIVATE BRAID::braid)
if(TARGET BRAID::static)
  add_executable(consumer_static ../../examples/encode.c)
  target_link_libraries(consumer_static PRIVATE BRAID::static)
  add_library(consumer_plugin SHARED plugin.c)
  target_link_libraries(consumer_plugin PRIVATE BRAID::static)
  target_link_options(consumer_plugin PRIVATE "LINKER:--no-undefined")
endif()

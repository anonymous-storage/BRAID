function(braid_private target)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/include")
  target_compile_options(${target} PRIVATE
    "-ffile-prefix-map=${PROJECT_SOURCE_DIR}=."
    "-fdebug-prefix-map=${PROJECT_SOURCE_DIR}=."
    "-ffile-prefix-map=${PROJECT_BINARY_DIR}=./build"
    "-fdebug-prefix-map=${PROJECT_BINARY_DIR}=./build")
endfunction()

set(core_sources src/core/gf256.cc src/core/rs_matrix.cc src/core/decomposition.cc
  src/runtime/context.cc src/runtime/plan.cc src/runtime/topology.cc
  src/runtime/thread_pool.cc src/runtime/c_api.cc src/runtime/cpu_features.cc
  src/kernels/grape_scalar.cc src/kernels/backend_dispatch.cc)
set(extra_objects)
set(backend_definitions)
if(BRAID_ENABLE_X86 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  check_cxx_compiler_flag("-mavx2" has_avx2)
  check_cxx_compiler_flag("-mgfni" has_gfni)
  check_cxx_compiler_flag("-mavx512bw" has_avx512)
  if(has_avx2)
    add_library(braid_avx2 OBJECT src/kernels/avx2_kernels.cc)
    braid_private(braid_avx2)
    target_compile_options(braid_avx2 PRIVATE -mavx2 -fomit-frame-pointer)
    list(APPEND extra_objects $<TARGET_OBJECTS:braid_avx2>)
    list(APPEND backend_definitions BRAID_HAVE_AVX2=1)
  endif()
  if(has_avx2 AND has_avx512)
    add_library(braid_avx512 OBJECT src/kernels/avx512_kernels.cc)
    braid_private(braid_avx512)
    target_compile_options(braid_avx512 PRIVATE -mavx2 -mavx512f -mavx512bw -mavx512vl -fomit-frame-pointer)
    list(APPEND extra_objects $<TARGET_OBJECTS:braid_avx512>)
    list(APPEND backend_definitions BRAID_HAVE_AVX512=1)
    if(has_gfni)
      add_library(braid_avx512_gfni OBJECT src/kernels/avx512_kernels.cc)
      braid_private(braid_avx512_gfni)
      target_compile_options(braid_avx512_gfni PRIVATE -mavx2 -mavx512f -mavx512bw -mavx512vl -mgfni -fomit-frame-pointer)
      target_compile_definitions(braid_avx512_gfni PRIVATE BRAID_AVX512_GFNI_OBJECT=1)
      list(APPEND extra_objects $<TARGET_OBJECTS:braid_avx512_gfni>)
      list(APPEND backend_definitions BRAID_HAVE_AVX512_GFNI=1)
    endif()
    if(BRAID_ENABLE_AVX512_PLAN_JIT)
      file(GLOB asmjit_sources CONFIGURE_DEPENDS
        third_party/asmjit/asmjit/core/*.cpp third_party/asmjit/asmjit/support/*.cpp
        third_party/asmjit/asmjit/x86/*.cpp)
      add_library(braid_asmjit OBJECT ${asmjit_sources})
      braid_private(braid_asmjit)
      target_include_directories(braid_asmjit PRIVATE third_party/asmjit)
      target_compile_definitions(braid_asmjit PRIVATE ASMJIT_STATIC ASMJIT_NO_AARCH64 ASMJIT_NO_FOREIGN)
      list(APPEND extra_objects $<TARGET_OBJECTS:braid_asmjit>)
      list(APPEND core_sources src/runtime/avx512_plan_jit.cc)
      list(APPEND backend_definitions BRAID_ENABLE_AVX512_PLAN_JIT=1 ASMJIT_STATIC)
      target_compile_definitions(braid_avx512 PRIVATE BRAID_ENABLE_AVX512_PLAN_JIT=1)
      if(TARGET braid_avx512_gfni)
        target_compile_definitions(braid_avx512_gfni PRIVATE BRAID_ENABLE_AVX512_PLAN_JIT=1)
      endif()
    endif()
  endif()
endif()
# The frozen NEON implementation is archived but deliberately not enabled.
# The later native ARM fork has a different planner and is not merged here.
add_library(braid_core OBJECT ${core_sources})
braid_private(braid_core)
target_include_directories(braid_core PRIVATE third_party/asmjit)
target_compile_definitions(braid_core PRIVATE BRAID_BUILDING_LIBRARY=1 ${backend_definitions})
target_link_libraries(braid_core PRIVATE Threads::Threads)

foreach(kind IN ITEMS shared static)
  string(TOUPPER "${kind}" upper_kind)
  if(BRAID_BUILD_${upper_kind})
    add_library(braid_${kind} ${upper_kind} $<TARGET_OBJECTS:braid_core> ${extra_objects})
    add_library(BRAID::${kind} ALIAS braid_${kind})
    set_target_properties(braid_${kind} PROPERTIES OUTPUT_NAME braid EXPORT_NAME ${kind}
      VERSION ${PROJECT_VERSION} SOVERSION 0 LINKER_LANGUAGE CXX)
    target_include_directories(braid_${kind} INTERFACE
      $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(braid_${kind} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if(kind STREQUAL "shared")
      target_link_options(braid_shared PRIVATE "LINKER:--no-undefined"
        "LINKER:--version-script=${PROJECT_SOURCE_DIR}/cmake/braid.map")
      set_property(TARGET braid_shared APPEND PROPERTY LINK_DEPENDS "${PROJECT_SOURCE_DIR}/cmake/braid.map")
    else()
      target_link_libraries(braid_static INTERFACE "$<$<LINK_LANGUAGE:C>:${BRAID_CXX_RUNTIME}>" "$<$<LINK_LANGUAGE:C>:m>")
    endif()
    install(TARGETS braid_${kind} EXPORT BRAIDTargets
      ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
  endif()
endforeach()
if(BRAID_BUILD_SHARED)
  set(default_target braid_shared)
else()
  set(default_target braid_static)
endif()

  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BRAID)
